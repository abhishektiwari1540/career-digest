import React, { useEffect, useState } from "react";

export function DailyChampionBlog() {
  const [blog, setBlog] = useState(null);

  useEffect(() => {
    fetch("https://career-digest.vercel.app/api/blogs/champion")
      .then((res) => res.json())
      .then((data) => { if (data.success && data.champion) setBlog(data.champion); });
  }, []);

  if (!blog) return null;
  return (
    <article className="daily-blog-card">
      <span className="badge">🏆 Daily Tech Champion Blog ({blog.seo_score}/100)</span>
      <h3>{blog.title}</h3>
      <p>{blog.tldr_summary}</p>
      <a href={\`https://www.abhishektiwari.online/blog/\${blog.slug}\`}>Read In-Depth Guide →</a>
    </article>
  );
}
